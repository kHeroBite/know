## Task 2: _구글AI개요검색() 함수 구현
**Files:** Modify `know-api/services/ai_analysis_service.py` — `_덕덕고검색` 함수 앞에 신규 함수 삽입
**Step 1:** 다음 함수를 `_덕덕고검색` 앞에 추가:
```python
async def _구글AI개요검색(keyword: str) -> dict:
    """Playwright 헤드리스 브라우저로 구글 검색 + AI개요 파싱."""
    try:
        from playwright.async_api import async_playwright
        async with async_playwright() as p:
            browser = await p.chromium.launch(headless=True, args=["--no-sandbox", "--disable-dev-shm-usage"])
            page = await browser.new_page()
            await page.set_extra_http_headers({
                "Accept-Language": "ko-KR,ko;q=0.9,en;q=0.8",
                "User-Agent": (
                    "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                    "AppleWebKit/537.36 (KHTML, like Gecko) "
                    "Chrome/120.0.0.0 Safari/537.36"
                ),
            })
            url = f"https://www.google.com/search?q={quote(keyword)}&hl=ko&gl=kr"
            logger.info("[AI분석][Google] Playwright 검색: url=%s", url)
            await page.goto(url, wait_until="domcontentloaded", timeout=15000)
            result = ""
            # AI 개요 선택자 순서 시도
            selectors = [
                "div[data-attrid='wa:/description'] span",
                "div.kno-rdesc span",
                "div[data-sncf='1'] div",
                "div.VwiC3b",
            ]
            for sel in selectors:
                el = await page.query_selector(sel)
                if el:
                    result = (await el.inner_text()).strip()
                    if result:
                        logger.info("[AI분석][Google] 선택자=%r 결과=%r", sel, result[:80])
                        break
            await browser.close()
            if not result:
                logger.info("[AI분석][Google] AI개요 없음 — '%s'", keyword)
                return {"result": None, "error": "AI개요 없음"}
            return {"result": result[:300], "error": None}
    except ImportError:
        logger.warning("[AI분석][Google] playwright 미설치 — DDG fallback")
        return {"result": None, "error": "playwright 미설치"}
    except Exception as e:
        logger.error("[AI분석][Google] 오류: %s", e)
        return {"result": None, "error": str(e)}
```

## Task 3: AI검색() 함수 수정
**Files:** Modify `know-api/services/ai_analysis_service.py` — `AI검색` 함수 내부

```python
async def AI검색(keyword: str) -> dict:
    """Playwright(구글) 우선 + DuckDuckGo fallback + Wikipedia 동시 검색."""
    import asyncio
    logger.info("[AI분석] 검색 시작: keyword=%r", keyword)
    prompt = 프롬프트로드()
    query = prompt.format(keyword=keyword)
    logger.info("[AI분석] 프롬프트 query=%r", query)
    # Playwright 구글 검색 우선 시도
    google_result = await _구글AI개요검색(keyword)
    if not google_result.get("result"):
        logger.info("[AI분석] Playwright 결과 없음 → DDG fallback")
        google_result = await _덕덕고검색(keyword)
    # Wikipedia는 항상 병렬 실행
    wiki_result = await _위키피디아검색(keyword)
    google_val = google_result.get("result") if google_result else None
    wiki_val = wiki_result.get("result") if wiki_result else None
    logger.info(
        "[AI분석] 최종 반환: google=%r, wikipedia=%r",
        google_val[:60] if google_val else None,
        wiki_val[:60] if wiki_val else None,
    )
    return {
        "google": google_val,
        "wikipedia": wiki_val,
        "query": query,
    }
```
